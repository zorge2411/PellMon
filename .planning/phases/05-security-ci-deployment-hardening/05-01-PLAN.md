         def hash_password(password, salt=None, iterations=100000):
             if salt is None:
                 salt = secrets.token_hex(16)
             derived = hashlib.pbkdf2_hmac('sha256', password.encode('utf-8'), salt.encode('utf-8'), iterations)
             return 'pbkdf2:sha256:%d$%s$%s' % (iterations, salt, derived.hex())

         def verify_password(stored_credential, provided_password):
             if not stored_credential or not provided_password:
                 return False
             if stored_credential.startswith('pbkdf2:sha256:'):
                 try:
                     parts = stored_credential.split('$')
                     iter_part = parts[0].split(':')[-1]
                     iterations = int(iter_part)
                     salt = parts[1]
                     stored_hash = parts[2]
                     derived = hashlib.pbkdf2_hmac('sha256', provided_password.encode('utf-8'), salt.encode('utf-8'), iterations)
                     return hmac.compare_digest(derived.hex(), stored_hash)
                 except Exception:
                     return False
             # Legacy plaintext fallback
             return hmac.compare_digest(stored_credential, provided_password)
